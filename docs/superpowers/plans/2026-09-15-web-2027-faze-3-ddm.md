V `src/app/opengraph-image.tsx` odstraň celou funkci `nacistFont` i obě volání `fetch` a nahraď je čtením souboru:

```ts
import { readFile } from 'node:fs/promises'
import { join } from 'node:path'

async function nacistFont(): Promise<ArrayBuffer | null> {
  try {
    const cesta = join(process.cwd(), 'public', 'fonts', 'bricolage-grotesque-variable.ttf')
    const buffer = await readFile(cesta)
    return buffer.buffer.slice(buffer.byteOffset, buffer.byteOffset + buffer.byteLength) as ArrayBuffer
  } catch {
    // Bez vlastního písma se náhled vykreslí systémovým — horší typografie je
    // pořád lepší než spadlý build, který zablokuje nasazení čehokoliv jiného.
    return null
  }
}
```

V `src/lib/payment-pricing.ts` přidej vedle ostatních `getTrusted*` funkcí:

```ts
import { getFocusModules } from './focus'

/**
 * Název tábora, jak se má objevit na faktuře a v e-mailu rodiči.
 * Skládá se z turnusu, ne z toho, co poslal klient — stejně jako cena,
 * kapacita, město i termín.
 */
export function getTrustedProgramName(termId: string, list: Turnus[] = TURNUSY): string {
  const turnus = resolveBookable(termId, list)
  const zamereni = getFocusModules(turnus.focus).map((m) => m.name)
  return zamereni.length > 0
    ? `Letní příměstský tábor — ${zamereni.join(', ')}`
    : 'Letní příměstský tábor'
}
```
